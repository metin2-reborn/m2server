# Unit tests: one executable per area, each registered with CTest.
function(m2_add_test name)
    cmake_parse_arguments(PARSE_ARGV 1 TEST "" "" "SOURCES;LIBRARIES")
    add_executable(${name} check_main.cpp ${TEST_SOURCES})
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/src)
    target_link_libraries(${name} PRIVATE ${TEST_LIBRARIES})
    add_test(NAME ${name} COMMAND ${name})
endfunction()

m2_add_test(log_tests SOURCES log_tests.cpp LIBRARIES metin2_log)
m2_add_test(utils_tests SOURCES utils_tests.cpp LIBRARIES metin2_core)
m2_add_test(grid_tests SOURCES grid_tests.cpp LIBRARIES metin2_common)
m2_add_test(config_tests SOURCES config_tests.cpp LIBRARIES metin2_config)
target_compile_definitions(config_tests PRIVATE M2_SOURCE_DIR="${PROJECT_SOURCE_DIR}")
m2_add_test(cipher_tests SOURCES cipher_tests.cpp ${PROJECT_SOURCE_DIR}/src/metin2/game/protocol/cipher.cpp LIBRARIES monocypher)
m2_add_test(sql_tests SOURCES sql_tests.cpp LIBRARIES metin2_sql)
m2_add_test(net_tests SOURCES net_tests.cpp LIBRARIES metin2_net)
m2_add_test(packet_sizes_tests SOURCES packet_sizes_tests.cpp LIBRARIES metin2_core)
